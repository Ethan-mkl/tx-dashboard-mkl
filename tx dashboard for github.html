function statusBadge(s){const m={Active:"green","At risk":"red","On hold":"amber"};return`<span class="badge ${m[s]||'blue'}"><span class="dot ${m[s]||'blue'}"></span>${s}</span>`}

function riskBadge(r){const m={low:"green",medium:"amber",high:"red"};return`<span class="badge ${m[r]}">${r.charAt(0).toUpperCase()+r.slice(1)}</span>`}

function scopeBadge(s){return s==="In-scope"?`<span class="badge green">In-scope</span>`:`<span class="badge amber">Scope-creep</span>`}


function renderContracts(tf,rf){

  let rows=contracts.filter(c=>(tf==="all"||c.team===tf)&&(rf==="all"||c.risk===rf));

  document.getElementById("contractBody").innerHTML=rows.map(c=>`<tr><td style="font-weight:500;color:#e8eaf0">${c.client}</td><td><span class="badge blue">${c.team.charAt(0).toUpperCase()+c.team.slice(1)}</span></td><td>${c.value}</td><td>${statusBadge(c.status)}</td><td>${scopeBadge(c.scope)}</td><td>${riskBadge(c.risk)}</td><td style="color:#6b7280">${c.renewal}</td></tr>`).join("");

}

function renderScope(tf,rf){

}

function renderPerf(tf){

  let t=teams.filter(t=>tf==="all"||t.id===tf);

  document.getElementById("perfCards").innerHTML=t.map(tm=>{

    const maxV=Math.max(...tm.trend),minV=Math.min(...tm.trend);

    const bars=tm.trend.map(v=>{const h=Math.round(8+((v-minV)/(maxV-minV+.001))*22);return`<div class="mini-bar" style="height:${h}px;background:${tm.color};opacity:.8"></div>`}).join("");

    const dir=tm.trend[5]>tm.trend[0]?"↑":"↓";

    const dirColor=tm.trend[5]>tm.trend[0]?"#34d399":"#f87171";

    return`<div class="perf-card"><div class="perf-header"><span class="perf-name">${tm.name}</span><span style="font-size:20px;font-weight:500;color:${tm.color}">${tm.score}</span></div><div style="display:flex;gap:8px;align-items:center;margin-bottom:6px"><span style="font-size:10px;color:#6b7280;width:60px">6-month</span><div class="mini-bars">${bars}</div><span style="font-size:13px;margin-left:4px;color:${dirColor}">${dir}</span></div><div class="tags-row">${tm.wins.map(w=>`<span class="win-tag">✓ ${w}</span>`).join("")}${tm.blocks.map(b=>`<span class="block-tag">⚠ ${b}</span>`).join("")}</div></div>`;

  }).join("");

}

let trendChart=null;

function renderTrendChart(tf){

  const ctx=document.getElementById("trendChart");

  if(!ctx)return;

  if(trendChart){trendChart.destroy();trendChart=null;}

  const filtered=teams.filter(t=>tf==="all"||t.id===tf);

  const dashes=[[],[6,3],[3,3],[8,3,2,3]];

  const datasets=filtered.map((t,i)=>({label:t.name,data:t.trend,borderColor:t.color,backgroundColor:"transparent",tension:.4,borderWidth:2,borderDash:dashes[i]||[],pointRadius:3,pointBackgroundColor:t.color}));

  document.getElementById("trendLegend").innerHTML=filtered.map(t=>`<span style="display:flex;align-items:center;gap:4px"><span style="width:10px;height:10px;border-radius:2px;background:${t.color}"></span>${t.name} — ${t.score}</span>`).join("");

  trendChart=new Chart(ctx,{type:"line",data:{labels:["Jan","Feb","Mar","Apr","May","Jun"],datasets},options:{responsive:true,maintainAspectRatio:false,plugins:{legend:{display:false}},scales:{x:{grid:{color:"rgba(255,255,255,.05)"},ticks:{color:"#6b7280",font:{size:11}}},y:{min:5,max:10,grid:{color:"rgba(255,255,255,.05)"},ticks:{color:"#6b7280",font:{size:11},stepSize:1}}}}});

}

  const rf=document.getElementById("riskFilter").value;

  renderContracts(tf,rf);

  renderScope(tf,rf);

  renderPerf(tf);

  renderTrendChart(tf);

}


function switchTab(name,el){

  document.querySelectorAll(".tab").forEach(t=>t.classList.remove("active"));

  document.querySelectorAll(".tab-panel").forEach(p=>p.classList.remove("active"));

  el.classList.add("active");

  document.getElementById("tab-"+name).classList.add("active");

  if(name==="performance"){setTimeout(()=>renderTrendChart(document.getElementById("teamFilter").value),50);}

}
